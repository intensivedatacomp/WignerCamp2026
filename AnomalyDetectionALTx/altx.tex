\section{The detailed calculation}

\begin{frame}{Step 1: Extracting the Laws}
  We define parameters \( r = 3, \; l = 2, \; k = 1 \). {\color{gray}(Their meaning will be discussed later.)}

  From \( a = [1, 1, 2, 3, 5] \) we extract the following segments of length \( r = 3 \):
  \begin{equation}
    [1, 1, 2], \; [1, 2, 3], \; [2, 3, 5].
  \end{equation}

  Then we embed them to \( l \times l = 2 \times 2 \) matrices:
  \begin{equation}
    \begin{bmatrix} 1 & 1 \\ 1 & 2 \end{bmatrix}, \quad
    \begin{bmatrix} 1 & 2 \\ 2 & 3 \end{bmatrix}, \quad
    \begin{bmatrix} 2 & 3 \\ 3 & 5 \end{bmatrix}.
  \end{equation}

  From those, the laws\footnote{eigenvectors corresponding to the smallest absolute eigenvalues} vectors which gives close to 0 when multiplying the embedded matrices, are:
  \begin{equation}
    \begin{bmatrix} -0.8507 \\ 0.5257 \end{bmatrix}, \quad
    \begin{bmatrix} -0.8507 \\ 0.5257 \end{bmatrix}, \quad
    \begin{bmatrix} -0.8507 \\ 0.5257 \end{bmatrix}.
  \end{equation}
\end{frame}

\begin{frame}{Step 1: Extracting the Laws (continued)}
  We combine the results into one matrix and repeat the process for all training instances (separately for each class).

  From class \textbf{a}, we obtain laws:
  \begin{equation}
    L_a = \begin{bmatrix} -0.8507 & -0.8507 & -0.8507 \\ 0.5257 & 0.5257 & 0.5257 \end{bmatrix}.
  \end{equation}

  From class \textbf{b}, we obtain laws:
  \begin{equation}
    L_b = \begin{bmatrix} -0.9571 & -0.8702 & -0.9085 \\ 0.2898 & 0.4927 & 0.4179 \end{bmatrix}.
  \end{equation}
\end{frame}

\begin{frame}{Step 2: Embedding the Test Instance}
  The test instance \( x = [2, 3, 5, 8, 13] \) is embedded as:
  \begin{equation}
    E = \begin{bmatrix} 2 & 3 \\ 3 & 5 \\ 5 & 8 \\ 8 & 13 \end{bmatrix}.
  \end{equation}
\end{frame}

\begin{frame}{Step 3: Projection by Laws}
  Multiplying with class \textbf{a} laws:
  \begin{equation}
    F_a = E L_a = \begin{bmatrix} -0.0344 & -0.0344 & -0.0344 \\ 0.0132 & 0.0132 & 0.0132 \\ -0.0050 & -0.0050 & -0.0050 \\ 0.0019 & 0.0019 & 0.0019 \end{bmatrix}.
  \end{equation}

  Multiplying with class \textbf{b} laws:
  \begin{equation}
    F_b = E L_b = \begin{bmatrix} -0.2898 & -0.0727 & -0.1563 \\ -0.2439 & -0.0252 & -0.1091 \\ -0.2615 & -0.0434 & -0.1272 \\ -0.2548 & -0.0365 & -0.1203 \end{bmatrix}.
  \end{equation}
\end{frame}

\begin{frame}{Step 4: Feature Calculation}
  Using the mean of all squared values:
  \begin{itemize}
    \item Class \textbf{a}: \( \text{mean} = 0.00035 \).
    \item Class \textbf{b}: \( \text{mean} = 0.029 \).
  \end{itemize}

  \textbf{Conclusion:} The test instance is closer to class \textbf{a}.
\end{frame}

\section{The Parameters \( r, l, k \)}

\begin{frame}{Meaning of Parameters \( r, l, k \)}
  \begin{itemize}
    \item \( r \): Length of the original sequence window used for feature extraction.
    \item \( l \): Embedding size, defines a symmetric \( l \times l \) matrix.
    \item \( 2l - 1 \): Number of equally spaced elements sampled from the window of length \( r \).
    \item \( k \): Step size (stride) used for shifting the window over the time series.
  \end{itemize}
\end{frame}

\begin{frame}{Example \(r, l, k\) values}
    Consider the following instance: \(x' = [2, 1, 5, 7, 17, 31, 65, 127, 257, 511]\).
    
    Let \(r=5, l= 2, \) and \(k=1\). Then \(2l-1=3\) and
    \vspace{10px}
    
    \(x'_1 = [\boxed{\boxed{2}, 1, \boxed{5}, 7, \boxed{17}}, 31, 65, 127, 257, 511].\)
    \(x'_2 = [2, \boxed{\boxed{1}, 5, \boxed{7}, 17, \boxed{31}}, 65, 127, 257, 511].\)
    \(x'_2 = [2, 1, \boxed{\boxed{5}, 7, \boxed{17}, 31, \boxed{65}}, 127, 257, 511].\)
    \(x'_3 = [2, 1, 5, \boxed{\boxed{7}, 17, \boxed{31}, 65, \boxed{127}}, 257, 511].\)
    \(x'_4 = [2, 1, 5, 7, \boxed{\boxed{17}, 31, \boxed{65}, 127, \boxed{257}}, 511].\)
    \(x'_5 = [2, 1, 5, 7, 17, \boxed{\boxed{31}, 65, \boxed{127}, 257, \boxed{511}}].\)
    
\end{frame}

\section{Anomaly Detection}

\begin{frame}{Anomaly Detection as a Special Case of Classification}
  Anomaly detection is classification with \textbf{one class}: the normal class.
  \begin{itemize}
    \item Laws are extracted from \textbf{normal} data only.
    \item Instead of one scalar per class, we compute one score \textbf{per embedded window}, giving a score \emph{vector}.
    \item \textbf{Peaks} in the score vector mark the anomaly locations.
  \end{itemize}
\end{frame}

\begin{frame}{Step 1: Laws from Normal Data}
  Normal data: \(a = [1, 1, 2, 3, 5, 8, 13]\), parameters \(r=3,\; l=2,\; k=1\).
  \begin{equation}
    L = \begin{bmatrix}
      -0.8507 & -0.8507 & -0.8507 & -0.8507 & -0.8507 \\
       0.5257 &  0.5257 &  0.5257 &  0.5257 &  0.5257
    \end{bmatrix}.
  \end{equation}
  All five laws are identical because every window follows the same Fibonacci rule.
\end{frame}

\begin{frame}{Step 2: Embedding the Test Instance}
  Test instance (anomalies at positions 0 and 3): \(x = [6, 1, 2, 8, 5, 8, 13]\).
  \begin{equation}
    E = \begin{bmatrix}
      6 & 1 \\ 1 & 2 \\ 2 & 8 \\ 8 & 5 \\ 5 & 8 \\ 8 & 13
    \end{bmatrix}.
  \end{equation}
  Row \(j\) is the pair \((x_j,\; x_{j+1})\).
\end{frame}

\begin{frame}{Step 3: Projection}
  \begin{equation}
    F = E \cdot L = {\footnotesize\begin{bmatrix}
      -0.753 & -0.753 & -0.753 & -0.753 & -0.753 \\
       0.090 &  0.090 &  0.090 &  0.090 &  0.090 \\
       0.304 &  0.304 &  0.304 &  0.304 &  0.304 \\
      -0.443 & -0.443 & -0.443 & -0.443 & -0.443 \\
      -0.005 & -0.005 & -0.005 & -0.005 & -0.005 \\
       0.002 &  0.002 &  0.002 &  0.002 &  0.002
    \end{bmatrix}}.
  \end{equation}
  Rows 0 and 3 have the largest absolute values.
\end{frame}

\begin{frame}{Step 4: Anomaly Scores}
  Average \(F^2\) over the laws (columns), keeping one score per row:
  \begin{equation}
    s_j = \text{mean}_m\!\left(F_{jm}^2\right).
  \end{equation}
  \begin{equation}
    s = [\underbrace{0.567}_{\text{anomaly}},\; 0.008,\; 0.092,\; \underbrace{0.196}_{\text{anomaly}},\; 0.000,\; 0.000].
  \end{equation}
  With threshold \(\tau = 0.15\): peaks at indices 0 and 3 are flagged as anomalous.

  The mean can be replaced by a \textbf{percentile} (e.g.\ 95th) for robustness.
\end{frame}

\begin{frame}{Index Mapping: Score Index to Original Position}
  The score vector has \(N - l + 1\) elements. Score \(s_j\) is computed from window
  \begin{equation}
    E_j = [x_j,\; x_{j+1},\; \dots,\; x_{j+l-1}],
  \end{equation}
  so a peak at index \(j\) indicates an anomaly at original position \(j\).

  Notebook example (\(l = 2\)):
  \begin{itemize}
    \item \(s_0 = 0.567 > \tau \;\Rightarrow\) anomaly at position \(0\). \checkmark
    \item \(s_3 = 0.196 > \tau \;\Rightarrow\) anomaly at position \(3\). \checkmark
  \end{itemize}
\end{frame}

\begin{frame}{Exercise: Index Mapping}
  \emph{altx} is applied to a time series of length \(N = 10\) with \(l = 3\).

  \vspace{0.5em}
  \textbf{Part (a).} How many elements does the anomaly score vector \(s\) have?

  \vspace{0.5em}
  \textbf{Part (b).} The score vector has peaks at indices \(j = 1\) and \(j = 6\).
  What are the corresponding anomaly positions in \(x\)?
\end{frame}

\begin{frame}{Exercise: Index Mapping -- Solutions}
  \textbf{Part (a).}
  \begin{equation}
    N - l + 1 = 10 - 3 + 1 = 8.
  \end{equation}

  \textbf{Part (b).} A peak at index \(j\) indicates an anomaly at position \(j\):
  \begin{itemize}
    \item Peak at \(j = 1 \;\Rightarrow\) anomaly at position 1.
    \item Peak at \(j = 6 \;\Rightarrow\) anomaly at position 6.
  \end{itemize}
\end{frame}

\begin{frame}{Summary of ALT Method}
  \begin{itemize}
    \item Extract subsequences and embed into symmetric matrices.
    \item Compute eigenvectors of smallest absolute eigenvalues — these are the laws.
    \item Multiply embedded test instances with class-specific laws.
    \item Compute summary statistics (e.g., mean of squared values).
    \item Classify by choosing the class that minimizes the statistic.
    \item By only taking statistic over the laws, the peaks represent anomalies.
  \end{itemize}
\end{frame}
