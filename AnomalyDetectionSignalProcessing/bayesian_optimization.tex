\section{Bayesian Optimisation}

\begin{frame}{The Hyperparameter Tuning Problem}
  \begin{itemize}
    \item A \textbf{hyperparameter} is a configuration value set before training (e.g.\ regularisation strength, number of trees, learning rate).
    \item We seek the vector \(\theta^*\) that maximises the cross-validation score \(f(\theta)\):
    \begin{equation}
      \theta^* = \arg\max_{\theta \in \Theta}\, f(\theta).
    \end{equation}
    \item Evaluating \(f(\theta)\) is expensive: it requires training and cross-validating the model for every candidate \(\theta\).
  \end{itemize}
\end{frame}

\begin{frame}{Search Strategies Compared}
  \begin{itemize}
    \item Three standard strategies differ in whether they exploit past evaluations.
  \end{itemize}
  \begin{center}
  \begin{tabular}{lcc}
    \hline
    \textbf{Strategy} & \textbf{Uses past info?} & \textbf{Cost scaling} \\
    \hline
    Grid search     & No  & Exponential in dims. \\
    Random search   & No  & Linear in dims. \\
    Bayesian opt.   & Yes & Sub-linear in dims. \\
    \hline
  \end{tabular}
  \end{center}
  \begin{itemize}
    \item Bayesian optimisation reuses information from all past evaluations to focus the search on the most promising regions of \(\Theta.\)
  \end{itemize}
\end{frame}

\begin{frame}{The Bayesian Approach}
  \begin{itemize}
    \item A \textbf{surrogate model} (e.g.\ a Gaussian Process) builds a probabilistic approximation \(\hat{f}(\theta)\) of the objective from all previously evaluated points.
    \item An \textbf{acquisition function} \(\alpha(\theta)\) quantifies how promising a candidate \(\theta\) is as the next evaluation point.
    \item At each iteration the next candidate is chosen as:
    \begin{equation}
      \theta_{n+1} = \arg\max_{\theta \in \Theta}\, \alpha\!\left(\theta;\, \{(\theta_i, f(\theta_i))\}_{i=1}^{n}\right).
    \end{equation}
    \item This balances \textbf{exploration} (uncertain regions) and \textbf{exploitation} (known good regions).
  \end{itemize}
\end{frame}

\section{BayesSearchCV}

\begin{frame}[fragile]{Function Signature}
  \texttt{BayesSearchCV} is a scikit-learn compatible wrapper for Bayesian hyperparameter search, documented at \href{https://scikit-optimize.github.io/stable/modules/generated/skopt.BayesSearchCV.html}{scikit-optimize.github.io}.
  \begin{lstlisting}
from skopt import BayesSearchCV

opt = BayesSearchCV(
    estimator,         # estimator to tune
    search_spaces,     # dict: param name -> search space
    n_iter=50,         # number of Bayesian evaluations
    cv=5,              # cross-validation
    scoring=None,      # evaluation metric
    n_jobs=1,          # used cores (-1 = all)
    random_state=None, # seed
    refit=True         # refit best model on all training data
)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{After-Fit Attributes}
  After calling \texttt{opt.fit(X\_train, y\_train)}, the following attributes are available:
  \begin{lstlisting}
opt.best_params_    # best hyperparameter combination
opt.best_score_     # best mean cross-validation score
opt.best_estimator_ # estimator refitted on all training data
opt.cv_results_     # full history of all evaluations
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{best\_estimator\_} is ready to use for prediction on new data.
    \item \texttt{cv\_results\_['params']} and \texttt{cv\_results\_['mean\_test\_score']} list every configuration tried together with its score.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{The \texttt{Real} and \texttt{Integer} Search Spaces}
  \begin{lstlisting}
from skopt.space import Real, Integer

# Continuous parameter on a linear scale
Real(low=0.01, high=10.0, prior='uniform')

# Continuous parameter on a logarithmic scale
Real(low=1e-4, high=1e2, prior='log-uniform')

# Integer parameter (e.g., number of trees)
Integer(low=10, high=500)
  \end{lstlisting}
  \begin{itemize}
    \item Each space object is defined by its lower bound, upper bound, and sampling distribution (\texttt{prior}).
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{The \texttt{Categorical} Space}
  \begin{lstlisting}
from skopt.space import Categorical

# Unordered discrete choices
Categorical(categories=['rbf', 'linear', 'poly'])

# A typical search_spaces dictionary combining all types:
search_spaces = {
    'kernel': Categorical(['rbf', 'linear']),
    'C':      Real(1e-2, 1e2, prior='log-uniform'),
    'n':      Integer(10, 200)
}
  \end{lstlisting}
  \begin{itemize}
    \item Each key in \texttt{search\_spaces} maps a parameter name to its search space object.
    \item \texttt{Categorical} is used for unordered discrete choices such as kernel types or activation functions.
  \end{itemize}
\end{frame}

\begin{frame}{The \texttt{log-uniform} Prior}
  \begin{itemize}
    \item With \texttt{prior='uniform'}, samples are spread evenly on a linear scale across \([a, b].\)
    \item For wide ranges this concentrates most samples near the upper bound, leaving the lower end undersampled.
    \item With \texttt{prior='log-uniform'}, the logarithm of the parameter is uniform:
    \begin{equation}
      \log \theta \sim \text{Uniform}(\log a,\; \log b).
    \end{equation}
    \item This assigns \emph{equal probability to every order of magnitude.}
  \end{itemize}
\end{frame}

\begin{frame}{When to Use \texttt{log-uniform}}
  \begin{example}[\texttt{Real(1e-2, 1e2, prior='log-uniform')}]
    Equal probability is assigned to \([10^{-2}, 10^{-1}]\), \([10^{-1}, 10^{0}]\), \([10^{0}, 10^{1}]\), and \([10^{1}, 10^{2}].\)
  \end{example}
  \begin{itemize}
    \item Use \texttt{'log-uniform'} for parameters where relative differences matter more than absolute ones: regularisation strengths (\(C\), \(\lambda\)), learning rates (\(\eta\)), and kernel widths (\(\gamma\)).
    \item Use \texttt{'uniform'} (the default) for parameters where the effect is linear in scale, such as mixture weights or proportions.
  \end{itemize}
\end{frame}

\begin{frame}{\texttt{n\_iter} and \texttt{n\_jobs}}
  \begin{itemize}
    \item \textbf{\texttt{n\_iter}}: number of hyperparameter configurations to evaluate.
    \begin{itemize}
      \item Total model fits required: \(n\_\text{iter} \times k_\text{cv}.\)
      \item More iterations yield a better optimum but increase wall-clock time.
      \item A reasonable starting point for moderate search spaces is \(30\)--\(50.\)
    \end{itemize}
    \item \textbf{\texttt{n\_jobs}}: number of parallel jobs to run.
    \begin{itemize}
      \item Set to \(-1\) to use all available CPU cores.
      \item Can significantly reduce wall-clock time for large \texttt{n\_iter} or expensive estimators.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{\texttt{cv}, \texttt{scoring}, and \texttt{random\_state}}
  \begin{itemize}
    \item \textbf{\texttt{cv}}: cross-validation strategy.
    \begin{itemize}
      \item An integer \(k\) applies stratified \(k\)-fold CV (classification) or \(k\)-fold (regression).
      \item A sklearn splitter (e.g.\ \texttt{StratifiedKFold}) gives finer control.
    \end{itemize}
    \item \textbf{\texttt{scoring}}: the metric used to rank configurations.
    \begin{itemize}
      \item Classification: \texttt{'accuracy'}, \texttt{'f1'}, \texttt{'roc\_auc'}.
      \item Regression: \texttt{'r2'}, \texttt{'neg\_mean\_squared\_error'}.
      \item \texttt{None} uses the estimator's default \texttt{score()} method.
    \end{itemize}
    \item \textbf{\texttt{random\_state}}: integer seed that ensures reproducible runs.
  \end{itemize}
\end{frame}

\section{Single-Parameter Search}

\begin{frame}[fragile]{Setup: Imports and Data}
  We optimise the regularisation strength \(C\) of \texttt{LogisticRegression} on the breast cancer dataset.
  Distance-based models such as \texttt{LogisticRegression} require scaled features to converge reliably.
  \begin{lstlisting}
from skopt import BayesSearchCV
from skopt.space import Real
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.datasets import load_breast_cancer

X, y = load_breast_cancer(return_X_y=True)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Creating and Fitting the Search}
  \begin{lstlisting}
pipe = Pipeline([
    ('scaler', StandardScaler()),
    ('clf',    LogisticRegression(max_iter=1000))
])
opt = BayesSearchCV(
    estimator=pipe,
    search_spaces={'clf__C': Real(1e-3, 1e2, prior='log-uniform')},
    n_iter=20, cv=5, scoring='accuracy', random_state=42
)
opt.fit(X, y)
  \end{lstlisting}
  \begin{itemize}
    \item The key \texttt{'clf\_\_C'} follows the sklearn Pipeline convention: \texttt{step\_name\_\_param\_name.}
    \item The objective function is \(f(C) = \text{mean 5-fold CV accuracy of the pipeline}(C).\)
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Accessing the Best Result}
  \begin{lstlisting}
print(f"Best C:     {opt.best_params_['clf__C']:.4f}")
print(f"Best score: {opt.best_score_:.4f}")
# Example output:
# Best C:     0.3142
# Best score: 0.9754

# Use the best pipeline for prediction on new data
y_pred = opt.best_estimator_.predict(X_new)
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{best\_estimator\_} is the full \texttt{Pipeline} (scaler + classifier) already refitted on the full training set with the optimal \(C.\)
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Inspecting All Evaluated Configurations}
  \begin{lstlisting}
for params, score in zip(
        opt.cv_results_['params'],
        opt.cv_results_['mean_test_score']):
    print(params, f"-> {score:.4f}")
# Example output:
# {'clf__C': 0.0132} -> 0.9173
# {'clf__C': 4.5871} -> 0.9736
# {'clf__C': 0.3142} -> 0.9754
  \end{lstlisting}
  \begin{itemize}
    \item Iterating \texttt{cv\_results\_} is useful for plotting the optimisation history and checking convergence.
  \end{itemize}
\end{frame}

\section{Multi-Parameter Search}

\begin{frame}[fragile]{Multi-Parameter Search: Problem Setup}
  \begin{itemize}
    \item We tune three hyperparameters of an \texttt{SVC} simultaneously: \(C\), \(\gamma\), and the kernel type.
  \end{itemize}
  \begin{lstlisting}
from skopt import BayesSearchCV
from skopt.space import Real, Categorical
from sklearn.svm import SVC
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.datasets import load_breast_cancer

X, y = load_breast_cancer(return_X_y=True)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Multi-Parameter Search: The Call}
  \begin{lstlisting}
pipe = Pipeline([
    ('scaler', StandardScaler()),
    ('svc',    SVC())
])
opt = BayesSearchCV(
    estimator=pipe,
    search_spaces={
        'svc__C':      Real(1e-2, 1e2,  prior='log-uniform'),
        'svc__gamma':  Real(1e-4, 1e-1, prior='log-uniform'),
        'svc__kernel': Categorical(['rbf', 'poly'])
    },
    n_iter=40, cv=5, scoring='accuracy', random_state=42
)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Multi-Parameter Search: Fitting and Results}
  \begin{lstlisting}
opt.fit(X, y)
print(opt.best_params_)
print(f"Best score: {opt.best_score_:.4f}")
# Example output:
# {'svc__C': 12.4, 'svc__gamma': 0.0012, 'svc__kernel': 'rbf'}
# Best score: 0.9894
  \end{lstlisting}
  \begin{itemize}
    \item The objective function is \(f(C, \gamma, \text{kernel}) = \text{mean CV accuracy of the pipeline}(C, \gamma, \text{kernel}).\)
  \end{itemize}
\end{frame}

\begin{frame}{Why Use Bayesian Search?}
  \begin{itemize}
    \item The surrogate model captures correlations between parameters automatically, so the search focuses on jointly promising regions.
    \item \textbf{Grid search:} a \(10 \times 10 \times 2\) grid requires \(200\) evaluations; Bayesian search finds a comparable optimum in \(40.\)
    \item The advantage grows with the number of hyperparameters: grid search cost scales exponentially while Bayesian search scales much more slowly.
    \item \texttt{log-uniform} priors are used for both \(C\) and \(\gamma\) since their effects are relative, spanning several orders of magnitude.
  \end{itemize}
\end{frame}

\begin{frame}{Grid vs Bayesian: Scaling}
  \begin{itemize}
    \item The table shows evaluations required by each method, using \(10\) grid points per parameter.
  \end{itemize}
  \begin{center}
  \begin{tabular}{ccc}
    \hline
    \textbf{Parameters} & \textbf{Grid (10 pts each)} & \textbf{Bayesian} \\
    \hline
    2 & \(10^2 = 100\) & \(\sim 30\) \\
    4 & \(10^4 = 10\,000\) & \(\sim 50\) \\
    6 & \(10^6 = 1\,000\,000\) & \(\sim 100\) \\
    \hline
  \end{tabular}
  \end{center}
\end{frame}

\end{frame}

\begin{frame}[fragile]{Exercise 3 -- Debug the Code (1/2)}
  The following \texttt{BayesSearchCV} call contains \textbf{three bugs}.
  Find and fix all of them.
  \begin{lstlisting}
from skopt import BayesSearchCV
from skopt.space import Real
from sklearn.svm import SVC
from sklearn.datasets import load_breast_cancer

X, y = load_breast_cancer(return_X_y=True)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Exercise 3 -- Debug the Code (2/2)}
  \begin{lstlisting}
opt = BayesSearchCV(
    estimator=SVC, # line A
    search_spaces={
        'C':     Real(1e2, 1e-2, prior='log-uniform'), # line B
        'gamma': Real(1e-4, 1e-1, prior='log-uniform'),
    },
    n_iter=20,
    cv=5,
    random_state=42
)
opt.fit(X, y)
print(opt.best_score) # line C
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Exercise 3 -- Solution (1/2)}
  \begin{enumerate}
    \item \textbf{Line A}: \texttt{SVC} is a class, not an instance.
    The \texttt{estimator} argument requires an instantiated object:
    \begin{lstlisting}
estimator=SVC()
    \end{lstlisting}
    \item \textbf{Line B}: \texttt{Real(1e2, 1e-2, ...)} has \(low > high\) (\(100 > 0.01\)), which is invalid.
    The bounds must satisfy \(low \leq high\):
    \begin{lstlisting}
'C': Real(1e-2, 1e2, prior='log-uniform')
    \end{lstlisting}
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Exercise 3 -- Solution (2/2)}
  \begin{enumerate}
    \setcounter{enumi}{2}
    \item \textbf{Line C}: \texttt{best\_score} is not a valid attribute.
    All fitted attributes in scikit-learn end with a trailing underscore:
    \begin{lstlisting}
print(opt.best_score_)
    \end{lstlisting}
  \end{enumerate}
  \begin{itemize}
    \item As a general rule, any attribute computed during \texttt{fit()} in scikit-learn ends with \texttt{\_} (e.g.\ \texttt{best\_params\_}, \texttt{best\_score\_}, \texttt{best\_estimator\_}, \texttt{cv\_results\_}).
  \end{itemize}
\end{frame}
